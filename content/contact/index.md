  function updateNepalClock() {
    const timeDisplay = document.getElementById("npt-time");
    if (!timeDisplay) return;
    const now = new Date();
    const utc = now.getTime() + (now.getTimezoneOffset() * 60000);
    const nepalTime = new Date(utc + (3600000 * 5.75));
    timeDisplay.textContent = nepalTime.toLocaleTimeString("en-US", {
      hour: "2-digit",
      minute: "2-digit",
      second: "2-digit",
      hour12: true
    });
  }
  updateNepalClock();
  setInterval(updateNepalClock, 1000);

  function copyEmailAddress() {
    const email = "workwithrisaav@gmail.com";
    navigator.clipboard.writeText(email).then(() => {
      const btnText = document.getElementById("copy-email-text");
      if (btnText) {
        const originalText = btnText.textContent;
        btnText.textContent = "Copied! ✓";
        setTimeout(() => {
          btnText.textContent = originalText;
        }, 2000);
      }
    }).catch(err => {
      window.location.href = "mailto:" + email;
    });
  }

  function focusComposer(topic) {
    const subjectSelect = document.getElementById("form-subject");
    if (subjectSelect && topic) {
      for (let i = 0; i < subjectSelect.options.length; i++) {
        if (subjectSelect.options[i].value.includes(topic)) {
          subjectSelect.selectedIndex = i;
          break;
        }
      }
    }
    const composer = document.getElementById("composer-section");
    if (composer) {
      composer.scrollIntoView({ behavior: "smooth" });
      const nameInput = document.getElementById("form-name");
      if (nameInput) nameInput.focus();
    }
  }

  function sendEmailDraft() {
    const name = document.getElementById("form-name").value.trim();
    const email = document.getElementById("form-email").value.trim();
    const subjectTopic = document.getElementById("form-subject").value;
    const message = document.getElementById("form-message").value.trim();

    const subject = encodeURIComponent(`[Inquiry] ${subjectTopic} — ${name}`);
    const body = encodeURIComponent(
      `Hello Rishav,\n\n${message}\n\n---\nName: ${name}\nEmail: ${email}\nTopic: ${subjectTopic}`
    );

    window.location.href = `mailto:workwithrisaav@gmail.com?subject=${subject}&body=${body}`;
  }

  function copyDraftToClipboard() {
    const name = document.getElementById("form-name").value.trim() || "[Your Name]";
    const email = document.getElementById("form-email").value.trim() || "[Your Email]";
    const subjectTopic = document.getElementById("form-subject").value;
    const message = document.getElementById("form-message").value.trim() || "[Your Message]";

    const fullDraft = `To: workwithrisaav@gmail.com\nSubject: [Inquiry] ${subjectTopic} — ${name}\n\nHello Rishav,\n\n${message}\n\n---\nName: ${name}\nEmail: ${email}\nTopic: ${subjectTopic}`;

    navigator.clipboard.writeText(fullDraft).then(() => {
      const toast = document.getElementById("composer-toast");
      if (toast) {
        toast.style.display = "inline-block";
        setTimeout(() => {
          toast.style.display = "none";
        }, 3000);
      }
    });
  }
</script>
